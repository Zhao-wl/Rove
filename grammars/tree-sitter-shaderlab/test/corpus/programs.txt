==================
CG and HLSL include and program blocks
==================

Shader "Example/Program Blocks" {
    CGINCLUDE
    float cgSupport() { return 1.0; }
    ENDCG
    HLSLINCLUDE
    float hlslSupport() { return 2.0; }
    ENDHLSL
    SubShader {
        Pass {
            CGPROGRAM
            #pragma vertex vert
            float4 vert(float4 position : POSITION) : SV_POSITION { return position; }
            ENDCG
        }
        Pass {
            HLSLPROGRAM
            #pragma fragment frag
            float4 frag() : SV_Target { return 1; }
            ENDHLSL // Closing comments are outside the injected fragment.
        }
    }
}
