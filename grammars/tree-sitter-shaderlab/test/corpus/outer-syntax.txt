==================
Properties, tags and render states
==================

Shader "Example/Properties" {
    Properties {
        [HDR] _Tint ("Tint", Color) = (1, 0.5, 0, 1)
        [Header(Cutout Settings)] _Cutoff ("Cutoff", RANGE(0, 1)) = 0.5
        _MainTex ("Texture", 2d) = "white" {}
    }
    Subshader {
        Tags { "Queue" = "Transparent" }
        Pass {
            CUll Back
            Zwrite Off
            blend SrcAlpha OneMinusSrcAlpha
            Color [_Tint]
            Stencil { Ref 2 Comp Always Pass Replace }
        }
    }
    Fallback "Diffuse"
    CustomEditor "Example.Editor"
}
